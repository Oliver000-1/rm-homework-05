cmake_minimum_required(VERSION 3.16)
project(rm_homework_05 VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(BUILD_LIVE_APP "Build the C++ camera/replay app" ON)
option(BUILD_CALIBRATION "Build the chessboard calibration tool" ON)
option(BUILD_ORT_SMOKE "Build headless ONNX fixture verification tool" ON)
option(ENABLE_MVS "Enable the installed Linux camera SDK" ON)
set(MVS_ROOT "/opt/MVS" CACHE PATH "Installed MVS SDK root")
set(ONNXRUNTIME_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/third_party/onnxruntime-linux-x64-1.31.0" CACHE PATH "Official CPU C++ runtime root")

enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_include_directories(core_tests PRIVATE include)
add_test(NAME ownership_geometry_decode_nms COMMAND core_tests)

if(BUILD_LIVE_APP OR BUILD_CALIBRATION)
    set(OPENCV_COMPONENTS core imgproc imgcodecs calib3d video)
    if(BUILD_LIVE_APP)
        list(APPEND OPENCV_COMPONENTS videoio highgui)
    endif()
    find_package(OpenCV 4.5.4 REQUIRED COMPONENTS ${OPENCV_COMPONENTS})
    add_library(armor_geometry src/geometry.cpp src/tracking.cpp)
    target_include_directories(armor_geometry PUBLIC include ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(armor_geometry PUBLIC ${OpenCV_LIBS})
    add_executable(geometry_tests tests/geometry_tests.cpp)
    target_link_libraries(geometry_tests PRIVATE armor_geometry)
    add_test(NAME calibration_pose_tracking COMMAND geometry_tests)
endif()
if(BUILD_CALIBRATION)
    add_executable(armor_calibrate src/calibrate.cpp)
    target_link_libraries(armor_calibrate PRIVATE armor_geometry)
endif()
if(BUILD_LIVE_APP OR BUILD_ORT_SMOKE)
    find_path(ORT_INCLUDE onnxruntime_cxx_api.h HINTS "${ONNXRUNTIME_ROOT}/include" NO_DEFAULT_PATH)
    find_library(ORT_LIBRARY NAMES onnxruntime HINTS "${ONNXRUNTIME_ROOT}/lib" NO_DEFAULT_PATH)
    if(NOT ORT_INCLUDE OR NOT ORT_LIBRARY)
        message(FATAL_ERROR "Run bash scripts/prepare_ubuntu.sh first, or set ONNXRUNTIME_ROOT to official C++ headers+CPU library")
    endif()
    add_library(armor_model src/onnx_model.cpp)
    target_include_directories(armor_model PUBLIC include PRIVATE "${ORT_INCLUDE}")
    target_link_libraries(armor_model PUBLIC "${ORT_LIBRARY}")
    get_filename_component(ORT_LIBRARY_DIR "${ORT_LIBRARY}" DIRECTORY)
endif()
if(BUILD_ORT_SMOKE)
    add_executable(ort_smoke tests/onnx_smoke.cpp)
    target_link_libraries(ort_smoke PRIVATE armor_model)
    add_test(NAME real_onnx_model_reference COMMAND ort_smoke
        "${CMAKE_CURRENT_SOURCE_DIR}/models/best.onnx"
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/frame180-input.f32"
        "${CMAKE_CURRENT_BINARY_DIR}/model-output.f32"
        "${CMAKE_CURRENT_BINARY_DIR}/model-detections.csv"
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/frame180-reference.csv")
    set_target_properties(ort_smoke PROPERTIES BUILD_RPATH "${ORT_LIBRARY_DIR}")
endif()
if(BUILD_LIVE_APP)
    find_package(Threads REQUIRED)
    add_executable(armor_live src/main.cpp src/camera.cpp src/recorder.cpp)
    target_include_directories(armor_live PRIVATE include ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(armor_live PRIVATE armor_model armor_geometry Threads::Threads)
    set_target_properties(armor_live PROPERTIES BUILD_RPATH "${ORT_LIBRARY_DIR}")
    if(ENABLE_MVS)
        if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
            message(FATAL_ERROR "USB camera build targets Linux. Use Ubuntu; host-only checks require -DENABLE_MVS=OFF")
        endif()
        find_path(MVS_INCLUDE MvCameraControl.h HINTS "${MVS_ROOT}/include" NO_DEFAULT_PATH)
        find_library(MVS_LIBRARY NAMES MvCameraControl HINTS "${MVS_ROOT}/lib/64" NO_DEFAULT_PATH)
        if(NOT MVS_INCLUDE OR NOT MVS_LIBRARY)
            message(FATAL_ERROR "Linux x86_64 MVS SDK not found; verify MVS_ROOT. Do not use Windows DLLs")
        endif()
        target_compile_definitions(armor_live PRIVATE ARMOR_WITH_MVS=1)
        target_include_directories(armor_live PRIVATE "${MVS_INCLUDE}")
        target_link_libraries(armor_live PRIVATE "${MVS_LIBRARY}")
        get_filename_component(MVS_LIBRARY_DIR "${MVS_LIBRARY}" DIRECTORY)
        set_property(TARGET armor_live APPEND PROPERTY BUILD_RPATH "${MVS_LIBRARY_DIR}")
    endif()
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(armor_live PRIVATE -Wall -Wextra -Wpedantic)
        target_compile_options(armor_model PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endif()
